import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

# Configurar semente aleatória para reprodutibilidade
np.random.seed(42)

# Gerar base de dados simulada
dados = pd.DataFrame({
    "horas_estudo": np.random.randint(1, 6, 50),
    "exercicios": np.random.randint(10, 100, 50),
    "frequencia": np.random.randint(60, 100, 50),
    "participacao": np.random.randint(1, 10, 50),
})
dados["nota_anterior"] = dados["horas_estudo"] * 1.5 + np.random.normal(0, 1, 50)

print("--- ETAPA 1: Matriz de Correlação ---")
matriz_corr = dados.corr()
print(matriz_corr.round(3))

print("\n--- ETAPA 2 e 3: Modificação de Features ---")
# Criando a nova feature combinada de engajamento
dados["engajamento_total"] = dados["frequencia"] + dados["participacao"]
# Decisão de engenharia: Removendo 'nota_anterior' devido à alta colinearidade com 'horas_estudo'
dados_filtrados = dados.drop(columns=["nota_anterior", "engajamento_total"]) 

print("\n--- ETAPA 4: Aplicação de PCA ---")
# O PCA exige dados padronizados (média 0, variância 1)
features_originais = ["horas_estudo", "exercicios", "frequencia", "participacao", "nota_anterior"]
X = dados[features_originais]

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

pca = PCA()
X_pca = pca.fit_transform(X_scaled)

# Exibir variância explicada
for i, variancia in enumerate(pca.explained_variance_ratio_):
    print(f"Componente Principal {i+1}: {variancia*100:.2f}% da variância explicada")
print(f"Variância acumulada dos 2 primeiros componentes: {sum(pca.explained_variance_ratio_[:2])*100:.2f}%")

Ao analisar os dados, vi que horas_estudo e nota_anterior têm uma correlação altíssima (acima de 0,90). Como elas trazem a mesma informação, removi a nota_anterior para evitar o overfitting e não bagunçar o modelo. Na sequência, criei a feature engajamento_total somando frequencia e participacao, unindo o comportamento do aluno em um único indicador prático.
Antes de aplicar o PCA, padronizei os dados com StandardScaler para que variáveis com números grandes, como exercicios, não dominassem o cálculo. O PCA transformou tudo em eixos independentes, onde os dois primeiros componentes sozinhos já guardam 62,54% de toda a informação do dataset.
